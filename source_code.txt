            socket = io('http://localhost:5000', { timeout: 2000 });
            
            socket.on('connect', () => {
                document.getElementById('socket-dot').className = 'w-2 h-2 rounded-full bg-emerald-500 animate-pulse';
                document.getElementById('socket-text').textContent = 'Backend Connected';
                document.getElementById('connection-status').className = 'hidden sm:flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[11px] font-semibold bg-emerald-50 text-emerald-700 border border-emerald-200';
            });

            socket.on('connect_error', () => {
                document.getElementById('socket-dot').className = 'w-2 h-2 rounded-full bg-amber-500';
                document.getElementById('socket-text').textContent = 'Offline Mode (Local Storage)';
            });

            socket.on('queueUpdate', (serverState) => {
                if (serverState && serverState.tokens) {
                    // Sync state from backend server
                    appState.currentServingNum = parseInt(serverState.counters[0]?.currentToken?.replace('A-', '') || appState.currentServingNum);
                    renderQueueTracker();
                    renderAdminTable();
                }
            });
        } catch(e) {
            console.log("Socket connection note: running standalone mode");
        }

        function saveState() {
            try {
                localStorage.setItem('queueless_state', JSON.stringify(appState));
            } catch (e) {

            }
        }

        function loadState() {
            try {
                const saved = localStorage.getItem('queueless_state');
                if (saved) {
                    appState = JSON.parse(saved);
                }
            } catch (e) {
                console.error('LocalStorage read error:', e);
            }
        }

        function resetDemoState() {

            saveState();
            switchTab('landing');

        // Voice Announcement using Web Speech API
        function speakAnnouncement(tokenStr, counterName) {
            if ('speechSynthesis' in window) {
                window.speechSynthesis.cancel();
                const cleanToken = tokenStr.replace('-', ' ');
                const text = `Token ${cleanToken}, please proceed to ${counterName}.`;
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.rate = 0.95;
                utterance.pitch = 1.0;
                window.speechSynthesis.speak(utterance);
            }
        }

        // Tab Switching Logic
        function switchTab(tabId) {
            appState.activeTab = tabId;
            saveState();
            document.querySelectorAll('.view-section').forEach(el => el.classList.add('hidden'));
            document.getElementById(`view-${tabId}`).classList.remove('hidden');

            ['landing', 'wizard', 'queue', 'admin'].forEach(t => {
                const btn = document.getElementById(`nav-${t}`);
                if (btn) {
                    if (t === tabId) {
                        btn.className = 'px-3 py-2 rounded-lg text-sm font-medium text-brand-600 bg-brand-50 transition-all';
                    } else {
                        btn.className = 'px-3 py-2 rounded-lg text-sm font-medium text-slate-600 hover:text-brand-600 hover:bg-slate-50 transition-all';
                    }
                }
            });

            window.scrollTo({ top: 0, behavior: 'smooth' });
            lucide.createIcons();
        }

            } else if (org === 'Bank') {
                depts = ['Cash Deposit & Withdrawal', 'Loan Enquiry', 'Account Opening & KYC', 'Customer Support', 'Forex Services'];
            } else if (org === 'Govt Office') {
                depts = ['Document Verification', 'License & Permit Renewal', 'Tax Assessment', 'Grievance Redressal', 'General Information'];

                depts = ['General Inquiry', 'Customer Support', 'Billing Counter', 'Technical Assistance'];

        // Handle token generation from wizard (with backend API sync if available)
        async function handleGenerateToken(e) {
            e.preventDefault();
            const name = document.getElementById('user-name').value;
            const contact = document.getElementById('user-contact').value;
            const service = document.getElementById('department-select').value;
            const orgType = document.querySelector('input[name="orgType"]:checked').value;

            // Try sending to backend if available
            try {
                const res = await fetch('http://localhost:5000/api/tokens', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ service, organization: orgType, userName: name, phone: contact })
                });
                const data = await res.json();
                if (data && data.token) {
                    appState.currentUser.tokenStr = data.token.tokenNumber;
                    appState.currentUser.tokenNum = parseInt(data.token.tokenNumber.replace('A-', ''));
                }
            } catch(err) {
                // Fallback to local generation if backend is offline
                appState.currentUser.tokenNum++;
                appState.currentUser.tokenStr = `A-${String(appState.currentUser.tokenNum).padStart(3, '0')}`;
            }

            appState.currentUser.name = name;
            appState.currentUser.contact = contact;
            appState.currentUser.service = service;
            appState.currentUser.orgType = orgType;
            appState.currentUser.status = 'waiting';

            document.getElementById('nav-queue-badge').classList.remove('hidden');
            document.getElementById('feedback-section').classList.add('hidden');

            saveState();
            switchTab('queue');
            renderQueueTracker();
            showToast(`🎟️ Digital Token ${appState.currentUser.tokenStr} successfully generated!`);
        }

        // Render Live Queue Tracker Page
        function renderQueueTracker() {
            document.getElementById('tracker-service-title').textContent = `${appState.currentUser.orgType} — ${appState.currentUser.service}`;
            document.getElementById('tracker-visitor-name').textContent = appState.currentUser.name;
            document.getElementById('tracker-token-display').textContent = appState.currentUser.tokenStr;
            document.getElementById('tracker-serving-token').textContent = `A-${String(appState.currentServingNum).padStart(3, '0')}`;
            
            const peopleAhead = Math.max(0, appState.currentUser.tokenNum - appState.currentServingNum - 1);
            document.getElementById('tracker-people-ahead').textContent = peopleAhead;

            // Heuristic AI wait time calculation formula = people ahead * avg service time
            const estMinutes = Math.max(1, Math.round(peopleAhead * appState.avgServiceMinutes));
            document.getElementById('tracker-wait-time').textContent = `${estMinutes} min`;
            document.getElementById('tracker-counter').textContent = appState.currentUser.counter;

            const priorityBadge = document.getElementById('tracker-priority-badge');
            if (appState.currentUser.isPriority) {
                priorityBadge.classList.remove('hidden');
            } else {
                priorityBadge.classList.add('hidden');
            }

            const statusText = document.getElementById('tracker-status-text');
            if (appState.currentServingNum >= appState.currentUser.tokenNum) {
                appState.currentUser.status = 'completed';
                statusText.className = 'px-3 py-1 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800';
                statusText.textContent = 'Status: Completed / Served';
                document.getElementById('feedback-section').classList.remove('hidden');
            } else if (peopleAhead <= 2) {
                statusText.className = 'px-3 py-1 rounded-full text-xs font-bold bg-rose-100 text-rose-800 animate-pulse';
                statusText.textContent = 'Status: Almost Your Turn!';
            } else {
                statusText.className = 'px-3 py-1 rounded-full text-xs font-bold bg-amber-100 text-amber-800';
                statusText.textContent = 'Status: Waiting in Queue';
            }

            // Render Queue Flow Chain
            const chain = document.getElementById('queue-flow-chain');
            chain.innerHTML = '';

            let nodes = [
                appState.currentServingNum,
                appState.currentServingNum + 1,
                appState.currentServingNum + 2,
                appState.currentServingNum + 3,
                appState.currentUser.tokenNum
            ];
            nodes = [...new Set(nodes)].sort((a, b) => a - b);

            nodes.forEach((num, idx) => {
                const isServing = num === appState.currentServingNum;
                const isUser = num === appState.currentUser.tokenNum;
                const str = `A-${String(num).padStart(3, '0')}`;

                const nodeDiv = document.createElement('div');
                nodeDiv.className = `px-4 py-3 rounded-xl border flex flex-col items-center justify-center transition-all ${
                    isUser ? 'bg-brand-600 text-white border-brand-600 shadow-lg shadow-brand-500/30 scale-105 font-bold' :
                    isServing ? 'bg-brand-50 text-brand-700 border-brand-300 font-bold' : 'bg-slate-50 text-slate-700 border-slate-200'
                }`;
                nodeDiv.innerHTML = `
                    <span class="text-[9px] uppercase font-semibold opacity-75">${isUser ? 'YOUR TOKEN' : isServing ? 'SERVING' : 'WAITING'}</span>
                    <span class="text-base font-black">${str}</span>
                `;
                chain.appendChild(nodeDiv);

                if (idx < nodes.length - 1) {
                    const arrow = document.createElement('div');
                    arrow.className = 'text-slate-300 font-bold';
                    arrow.innerHTML = '→';
                    chain.appendChild(arrow);
                }
            });

            saveState();
        }

        function shareTokenLink() {
            const shareText = `Track my QueueLess token ${appState.currentUser.tokenStr} live! Currently serving A-${String(appState.currentServingNum).padStart(3, '0')}.`;
            if (navigator.clipboard) {
                navigator.clipboard.writeText(shareText).then(() => {
                    showToast('📋 Companion link copied to clipboard!');
                });
            } else {
                showToast(shareText, 'info');
            }
        }

        // Post-service feedback rating
        function submitRating(label, score) {
            appState.ratingsTotal++;
            appState.ratingsSum += score;
            appState.satisfactionScore = (appState.ratingsSum / appState.ratingsTotal).toFixed(1);

            document.getElementById('admin-metric-satisfaction').textContent = `${appState.satisfactionScore}/5`;
            document.getElementById('feedback-section').innerHTML = `
                <div class="w-12 h-12 rounded-full bg-emerald-100 text-emerald-600 flex items-center justify-center mx-auto mb-3">
                    <i data-lucide="heart" class="w-6 h-6"></i>
                </div>
                <h4 class="text-xl font-bold text-slate-900 mb-1">Thank you for your feedback!</h4>
                <p class="text-xs text-slate-500">You rated your experience as <strong class="text-brand-600">${label} (${score}/5)</strong>. Have a wonderful day!</p>
            `;
            lucide.createIcons();
            saveState();
            showToast('⭐ Thank you for rating QueueLess!');
        }

        // Admin Actions & Dashboard Logic
        function adminCallNext() {
            if (appState.isPaused) {
                showToast('⚠️ Counter is currently paused by staff.', 'warning');
                return;
            }
            appState.currentServingNum++;
            appState.completedCount++;

            const selectedCounter = document.getElementById('admin-counter-select').value;
            appState.currentUser.counter = selectedCounter;

            const servingStr = `A-${String(appState.currentServingNum).padStart(3, '0')}`;

            // Trigger Voice Announcement
            speakAnnouncement(servingStr, selectedCounter);

            // Proximity notification for user
            const diff = appState.currentUser.tokenNum - appState.currentServingNum;
            if (diff === 2) {
                showToast('🔔 Notification: Your turn is approaching! Only 2 tokens away.', 'info');
            } else if (diff === 0) {
                showToast(`🎉 Notification: It's your turn! Please proceed to ${selectedCounter}.`, 'success');
            }

            updateAdminUI();
            renderQueueTracker();
            renderAdminTable();
            showToast(`📢 Called ${servingStr} to ${selectedCounter}`);
        }

        function adminRecall() {
            const servingStr = `A-${String(appState.currentServingNum).padStart(3, '0')}`;
            const counter = document.getElementById('admin-counter-select').value;
            speakAnnouncement(servingStr, counter);
            showToast(`🔔 Recalling ${servingStr} to ${counter}!`);
        }

        function adminSkip() {
            appState.currentServingNum++;
            updateAdminUI();
            renderQueueTracker();
            renderAdminTable();
            showToast('⏭️ Token skipped and moved to next.');
        }

        function adminTogglePause() {
            appState.isPaused = !appState.isPaused;
            const btn = document.getElementById('admin-pause-btn');
            const badge = document.getElementById('admin-queue-status-badge');
            if (appState.isPaused) {
                btn.textContent = 'Resume Counter';
                btn.className = 'px-4 py-3 rounded-xl bg-emerald-600 hover:bg-emerald-500 text-white font-semibold text-sm transition-all';
                badge.className = 'px-3.5 py-1.5 rounded-xl text-xs font-bold bg-amber-100 text-amber-800 flex items-center gap-1.5';
                badge.innerHTML = '<span class="w-2 h-2 rounded-full bg-amber-500 animate-pulse"></span> Counter Paused';
                showToast('⏸️ Counter paused successfully.', 'warning');
            } else {
                btn.textContent = 'Pause Counter';
                btn.className = 'px-4 py-3 rounded-xl bg-amber-600 hover:bg-amber-500 text-white font-semibold text-sm transition-all';
                badge.className = 'px-3.5 py-1.5 rounded-xl text-xs font-bold bg-emerald-100 text-emerald-800 flex items-center gap-1.5';
                badge.innerHTML = '<span class="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span> Queue Active';
                showToast('▶️ Counter resumed.', 'success');
            }
            saveState();
        }

        function toggleTokenPriority(tokenNum) {
            const item = appState.queueList.find(q => q.id === tokenNum);
            if (item) {
                item.isPriority = !item.isPriority;
                if (item.id === appState.currentUser.tokenNum) {
                    appState.currentUser.isPriority = item.isPriority;
                }
                saveState();
                renderAdminTable();
                renderQueueTracker();
                showToast(`⚡ Priority status updated for ${item.str}`);
            }
        }

        function speakSpecificToken(tokenStr, counter) {
            speakAnnouncement(tokenStr, counter);
            showToast(`🔊 Speaking announcement for ${tokenStr}`);
        }

        function updateAdminUI() {
            document.getElementById('admin-metric-serving').textContent = `A-${String(appState.currentServingNum).padStart(3, '0')}`;
            const waiting = Math.max(0, 49 - appState.currentServingNum);
            document.getElementById('admin-metric-waiting').textContent = waiting;
            document.getElementById('admin-metric-completed').textContent = appState.completedCount;
            document.getElementById('admin-metric-satisfaction').textContent = `${appState.satisfactionScore}/5`;
            saveState();
        }

        function renderAdminTable() {
            const tbody = document.getElementById('admin-table-body');
            if (!tbody) return;
            tbody.innerHTML = '';

            appState.queueList.forEach(t => {
                if (t.id <= appState.currentServingNum) return;

                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50 transition-all';
                tr.innerHTML = `
                    <td class="p-4 font-bold text-brand-600">${t.str}</td>
                    <td class="p-4 font-medium text-slate-800">${t.name}</td>
                    <td class="p-4 text-slate-600">${t.service}</td>
                    <td class="p-4"><span class="px-2.5 py-1 rounded-full text-xs font-semibold bg-amber-50 text-amber-700">Waiting (${t.counter})</span></td>
                    <td class="p-4">${t.isPriority ? '<span class="px-2 py-0.5 rounded text-[10px] bg-red-100 text-red-700 font-bold uppercase">Priority Active</span>' : '<span class="px-2 py-0.5 rounded text-[10px] bg-slate-100 text-slate-600">Standard</span>'}</td>
                    <td class="p-4 text-right space-x-2">
                        <button onclick="speakSpecificToken('${t.str}', '${t.counter}')" class="p-2 rounded-lg bg-indigo-50 text-indigo-600 hover:bg-indigo-100 text-xs font-semibold transition-all" title="Speak Announcement">
                            <i data-lucide="volume-2" class="w-4 h-4 inline"></i>
                        </button>
                        <button onclick="toggleTokenPriority(${t.id})" class="px-3 py-1.5 rounded-lg bg-amber-50 text-amber-700 hover:bg-amber-600 hover:text-white text-xs font-semibold transition-all">
                            ${t.isPriority ? 'Remove Priority' : 'Set Priority'}
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
            lucide.createIcons();
        }

        // Custom Toast Notification System
        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            
            let bgCol = 'bg-slate-900 text-white';
            if (type === 'success') bgCol = 'bg-emerald-600 text-white';
            if (type === 'warning') bgCol = 'bg-amber-600 text-white';
            if (type === 'info') bgCol = 'bg-brand-600 text-white';

            toast.className = `pointer-events-auto px-5 py-3 rounded-2xl shadow-2xl font-semibold text-sm flex items-center gap-3 transition-all transform translate-y-2 opacity-0 ${bgCol}`;
            toast.innerHTML = `<i data-lucide="bell" class="w-4 h-4"></i> <span>${message}</span>`;
            
            container.appendChild(toast);
            lucide.createIcons();

            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 4000);
        }
